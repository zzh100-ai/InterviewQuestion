## 基础算法

### Q1: 实现斐波那契数列（三种写法）

```python
# 方式一：递归（O(2^n)，有大量重复计算）
def fib(n):
    if n <= 1:
        return n
    return fib(n - 1) + fib(n - 2)

# 方式二：记忆化递归（O(n)，空间 O(n)）
from functools import lru_cache

@lru_cache(maxsize=None)
def fib_memo(n):
    if n <= 1:
        return n
    return fib_memo(n - 1) + fib_memo(n - 2)

# 方式三：迭代（O(n)，空间 O(1)，最优）
def fib_iter(n):
    if n <= 1:
        return n
    a, b = 0, 1
    for _ in range(2, n + 1):
        a, b = b, a + b
    return b
```

### Q2: 跳台阶问题（变态版）

普通版：每次只能跳 1 或 2 级，问 n 级台阶有多少种跳法 → 就是斐波那契。

变态版：每次能跳 1 ~ n 级 → `f(n) = 2^(n-1)`。

```python
def jump_floor(n):
    return 1 << (n - 1)  # 2^(n-1)

# 推导：f(n) = f(n-1) + f(n-2) + ... + f(1) = 2 * f(n-1) = 2^(n-1)
```

### Q3: 二分查找（手写，注意 mid 防溢出）

```python
def binary_search(arr, target):
    left, right = 0, len(arr) - 1
    while left <= right:
        mid = left + (right - left) // 2  # 防溢出，不用 (left+right)//2
        if arr[mid] == target:
            return mid
        elif arr[mid] < target:
            left = mid + 1
        else:
            right = mid - 1
    return -1
```

## 链表

### Q4: 单链表逆置

```python
class ListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next

def reverse_list(head):
    """三指针法"""
    prev = None
    curr = head
    while curr:
        nxt = curr.next  # 暂存下一个节点
        curr.next = prev # 反转指针
        prev = curr      # prev 前进
        curr = nxt       # curr 前进
    return prev

# 调试辅助
def list_to_array(head):
    res = []
    while head:
        res.append(head.val)
        head = head.next
    return res
```

### Q5: 链表成对调换

```python
# 1 → 2 → 3 → 4 变成 2 → 1 → 4 → 3
def swap_pairs(head):
    if not head or not head.next:
        return head
    new_head = head.next      # 第二个节点成为新头
    head.next = swap_pairs(new_head.next)  # head 的 next 指向下一对的头
    new_head.next = head       # 第二个节点指向第一个
    return new_head
```

### Q6: 交叉链表求交点

```python
def get_intersection_node(headA, headB):
    """长度对齐法：两个指针走完自己后去走对方的路，总路程相等时在交点相遇"""
    pA, pB = headA, headB
    while pA != pB:
        pA = pA.next if pA else headB
        pB = pB.next if pB else headA
    return pA  # 返回交点，或 None
```

## 二叉树

### Q7: 二叉树的前序、中序、后序遍历

```python
class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right

# 递归版
def preorder(root):
    """前序：根 → 左 → 右"""
    if not root: return []
    return [root.val] + preorder(root.left) + preorder(root.right)

def inorder(root):
    """中序：左 → 根 → 右"""
    if not root: return []
    return inorder(root.left) + [root.val] + inorder(root.right)

def postorder(root):
    """后序：左 → 右 → 根"""
    if not root: return []
    return postorder(root.left) + postorder(root.right) + [root.val]

# 迭代版（前序）
def preorder_iter(root):
    if not root: return []
    result, stack = [], [root]
    while stack:
        node = stack.pop()
        result.append(node.val)
        if node.right: stack.append(node.right)  # 右先入栈
        if node.left: stack.append(node.left)
    return result
```

### Q8: 二叉树的层次遍历（BFS）

```python
from collections import deque

def level_order(root):
    if not root: return []
    result, queue = [], deque([root])
    while queue:
        level = []
        for _ in range(len(queue)):
            node = queue.popleft()
            level.append(node.val)
            if node.left:
                queue.append(node.left)
            if node.right:
                queue.append(node.right)
        result.append(level)
    return result
# 输出：[[1], [2, 3], [4, 5, 6, 7]]
```

### Q9: 二叉树的最大深度

```python
def max_depth(root):
    if not root:
        return 0
    return 1 + max(max_depth(root.left), max_depth(root.right))

# BFS 版（适合层序遍历场景）
def max_depth_bfs(root):
    if not root: return 0
    depth, queue = 0, deque([root])
    while queue:
        depth += 1
        for _ in range(len(queue)):
            node = queue.popleft()
            if node.left: queue.append(node.left)
            if node.right: queue.append(node.right)
    return depth
```

### Q10: 根据前序 + 中序还原后序

```python
def rebuild(preorder, inorder):
    """
    前序第一个是根，中序中根左边是左子树、右边是右子树
    递归构建
    """
    if not preorder:
        return None
    root_val = preorder[0]
    root = TreeNode(root_val)
    idx = inorder.index(root_val)  # 根在中序中的位置
    root.left = rebuild(preorder[1:1+idx], inorder[:idx])
    root.right = rebuild(preorder[1+idx:], inorder[idx+1:])
    return root
```

## 排序与搜索

### Q11: 快速排序（手写）

```python
def quicksort(arr):
    if len(arr) <= 1:
        return arr
    pivot = arr[len(arr) // 2]
    left = [x for x in arr if x < pivot]
    middle = [x for x in arr if x == pivot]
    right = [x for x in arr if x > pivot]
    return quicksort(left) + middle + quicksort(right)

# 原地版（节省空间，面试常考）
def quicksort_inplace(arr, low=0, high=None):
    if high is None:
        high = len(arr) - 1
    if low < high:
        pivot_idx = partition(arr, low, high)
        quicksort_inplace(arr, low, pivot_idx - 1)
        quicksort_inplace(arr, pivot_idx + 1, high)

def partition(arr, low, high):
    pivot = arr[high]
    i = low - 1
    for j in range(low, high):
        if arr[j] <= pivot:
            i += 1
            arr[i], arr[j] = arr[j], arr[i]
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    return i + 1
```

时间复杂度：平均 O(n log n)，最坏 O(n²)（选到最小/最大元素作为 pivot 且数组已有序）。

### Q12: 合并两个有序列表

```python
def merge_two_lists(l1, l2):
    """双指针归并，O(n+m)"""
    result = []
    i = j = 0
    while i < len(l1) and j < len(l2):
        if l1[i] <= l2[j]:
            result.append(l1[i])
            i += 1
        else:
            result.append(l2[j])
            j += 1
    result.extend(l1[i:])
    result.extend(l2[j:])
    return result
```

### Q13: 杨氏矩阵查找

在一个每行每列都递增的矩阵中查找某个值。利用右上角的特殊位置：往左变小（列递减），往下变大（行递增）。

```python
def search_matrix(matrix, target):
    """Step-wise 线性搜索，O(m+n)"""
    if not matrix: return False
    m, n = len(matrix), len(matrix[0])
    row, col = 0, n - 1  # 从右上角开始
    while row < m and col >= 0:
        if matrix[row][col] == target:
            return True
        elif matrix[row][col] > target:
            col -= 1  # 大了往左走
        else:
            row += 1  # 小了往下走
    return False
```

### Q14: 找零问题（经典 DP）

给定不同面值的硬币和总金额，求用最少硬币数凑出目标金额。

```python
def coin_change(coins, amount):
    # dp[i] = 凑出金额 i 所需的最少硬币数
    dp = [float('inf')] * (amount + 1)
    dp[0] = 0
    for i in range(1, amount + 1):
        for coin in coins:
            if i >= coin:
                dp[i] = min(dp[i], dp[i - coin] + 1)
    return dp[amount] if dp[amount] != float('inf') else -1

print(coin_change([1, 2, 5], 11))  # 3（5+5+1）
```

### Q15: 去重（保留原始顺序）

```python
# 方式一：使用 set + 列表推导（保持顺序）
def dedupe_ordered(seq):
    seen = set()
    return [x for x in seq if not (x in seen or seen.add(x))]

# 方式二：dict.fromkeys（3.7+ 字典有序）
def dedupe_dict(seq):
    return list(dict.fromkeys(seq))

# 方式三：3.7+ 直接用 dict key 不重复
list(dict.fromkeys([3, 1, 2, 1, 3]))  # [3, 1, 2]
```

### Q16: 变位词判断

```python
# 两个字符串是否为变位词（字符相同，顺序不同）
def is_anagram(s1, s2):
    from collections import Counter
    return Counter(s1) == Counter(s2)

# 不用 Counter 的写法
def is_anagram2(s1, s2):
    if len(s1) != len(s2):
        return False
    count = {}
    for c in s1:
        count[c] = count.get(c, 0) + 1
    for c in s2:
        if c not in count:
            return False
        count[c] -= 1
        if count[c] < 0:
            return False
    return True
```

### Q17: 红黑树 vs AVL

| | AVL | 红黑树 |
|------|-----|--------|
| 平衡条件 | 严格（高度差 ≤ 1） | 宽松（最长路径 ≤ 2 × 最短路径） |
| 查询 | 更快（树更矮） | 稍慢 |
| 插入/删除 | 旋转次数多 | 最多 3 次旋转 |
| 适用场景 | 查询远多于写入 | 频繁插入删除 |

> Java HashMap 链表转树是红黑树（频繁插入），Linux CFS 调度器也用红黑树。
