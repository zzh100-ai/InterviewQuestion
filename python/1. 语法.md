# 1. 装饰器

>手工实现

```python
import time

def timing_decorator(func):
    def wrapper():
        start = time.time()
        result = func()          # 调用原函数
        end = time.time()
        print(f"函数 {func.__name__} 执行耗时: {end - start:.4f} 秒")
        return result
    return wrapper

def say_hello():
    print("Hello, World!")
    time.sleep(1)

# 手动装饰
say_hello = timing_decorator(say_hello)
say_hello()
```

>语法糖

```python
import time  
  
def timing_decorator(func):  
    def wrapper():  
        start = time.time()  
        result = func()          # 调用原函数  
        end = time.time()  
        print(f"函数 {func.__name__} 执行耗时: {end - start:.4f} 秒")  
        return result  
    return wrapper  
  
@timing_decorator  
def say_hello():  
    print("Hello, World!")  
    time.sleep(1)  
  
  
say_hello()
```
